      + placeComponent(c: Component, pos: Position) : boolean

      + getPositionOf(component : Component)

      + getChargedBatteries() : List<BatteryCompartment>

      - positions : Map<Player, Integer>

  abstract class Component {
      - connections : List<Connector>
      - componentDirection: Direction
      - damaged : boolean
      + initializeConnectors(ConnectorType n, ConnectorType e, ConnectorType s, ConnectorType w) void

  }

  enum ConnectorType{
    SINGLE
    DOUBLE
    UNIVERSAL
    NULL
  }

  class Connector {
      - direction : Direction
      - connectorType : ConnectorType
      - isCompatibleWith(Connector c1, Connector c2): boolean
  }

  class Cabin extends Component {
      - crew: List<Crew>
      - isStartingCabin : boolean
      - connectedLifeSupports : Set<LifeSupportSystemType>

      + canAddAlien(CrewType alienType) : boolean

  class ShieldGenerator extends Component {
  }
